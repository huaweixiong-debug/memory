```python
import json
from pathlib import Path

# 在 AIAgent 类中添加代理状态管理
class AIAgent:
    def __init__(self, ...):
        # ... 现有初始化代码 ...

        # 添加代理状态管理
        self.agent_state = {
            "current_agent": "hermes",
            "enabled": True
        }
        self._load_agent_state()

    def _load_agent_state(self):
        """从会话文件加载代理状态"""
        state_file = self.session_dir / "agent_state.json"
        if state_file.exists():
            try:
                with open(state_file, 'r') as f:
                    self.agent_state.update(json.load(f))
            except Exception:
                pass

    def _save_agent_state(self):
        """保存代理状态到会话文件"""
        state_file = self.session_dir / "agent_state.json"
        try:
            with open(state_file, 'w') as f:
                json.dump(self.agent_state, f, indent=2)
        except Exception:
            pass

    def _handle_use_command(self, message: str) -> str:
        """处理 /use 命令"""
        parts = message.strip().split()

        if len(parts) == 1:
            # 只输入 /use，显示当前代理
            current = self.agent_state["current_agent"]
            return f"当前活跃代理: {current}"

        if len(parts) >= 2:
            agent = parts[1].lower()
            supported_agents = ["hermes", "claude", "codex"]

            if agent in supported_agents:
                self.agent_state["current_agent"] = agent
                self._save_agent_state()
                return f"已切换到 {agent} 模式。后续消息将自动路由到 {agent}。"
            else:
                return f"不支持的代理: {agent}。支持的代理: {', '.join(supported_agents)}"

        return "用法: /use [hermes|claude|codex]"

    def _should_add_prefix(self, message: str) -> bool:
        """检查是否应该为消息添加前缀"""
        # 如果代理切换功能禁用，不添加前缀
        if not self.agent_state.get("enabled", True):
            return False

        # 如果当前代理是hermes，不添加前缀
        if self.agent_state["current_agent"] == "hermes":
            return False

        # 检查消息是否已有前缀
        stripped = message.strip().lower()
        if stripped.startswith("claude ") or stripped.startswith("claude:"):
            return False
        if stripped.startswith("codex ") or stripped.startswith("codex:"):
            return False

        return True

    def _add_agent_prefix(self, message: str) -> str:
        """为消息添加代理前缀"""
        current_agent = self.agent_state["current_agent"]
        return f"{current_agent} {message}"
```
