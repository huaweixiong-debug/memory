```powershell
$sshfs = "C:\Program Files\SSHFS-Win\bin\sshfs.exe"
$key = ($env:USERPROFILE -replace '\\','/') + "/.ssh/id_ed25519_ubuntu_codex2"
$knownHosts = ($env:USERPROFILE -replace '\\','/') + "/.ssh/known_hosts"

$env:Path = "C:\Program Files\SSHFS-Win\bin;$env:Path"

& $sshfs `
  "codex2@100.117.1.6:/home/codex2/projects" `

  -o "IdentityFile=$key" `

  -o "reconnect" `
  -o "ServerAliveInterval=30" `
  -o "idmap=user" `
  -o "umask=002" `
  -o "volname=UbuntuProjects"
```
